Goal: A sequence is split into lines of fixed length; positions and visible lines can be computed.

Functions (bodies are `todo!()` and must be replaced):

```
SequenceLayout::new(length, bases_per_line) -> Result<Self>   // bases_per_line == 0 → Err(Error::InvalidInput(..))
line_count() -> usize            // ceil(length / bases_per_line), for length 0: 0
line_of(position) -> usize       // position / bases_per_line
column_of(position) -> usize     // position % bases_per_line
line_start(line) -> usize        // line * bases_per_line
line_end(line) -> usize          // min(line_start + bases_per_line, length)
first_visible_line(scroll_y, line_height) -> usize
                                 // floor(scroll_y / line_height); for scroll_y <= 0 or line_height <= 0: 0
last_visible_line(&self, scroll_y, view_height, line_height) -> usize
                                 // floor((scroll_y + view_height) / line_height), capped at line_count() - 1
                                 // (for 0 lines: 0); for line_height <= 0: 0
bases_per_line_for(view_width, char_width, margin_width, group_size) -> usize
                                 // treat group_size 0 like 1
                                 // group width = (group_size + 1) * char_width
                                 // available = max(view_width - margin_width, 0)
                                 // groups = floor(available / group width), at least 1
                                 // result = groups * group_size
```

- `first_visible_line(250, 20)` = 12; `last_visible_line` for 100 lines, scroll 250, height 100, line height 20 → 17.

```rust
//! Common error type of the core.

use thiserror::Error;

/// Errors of the core. Messages are intended for users (German).
#[derive(Debug, Error, PartialEq)]
pub enum Error {
    /// Invalid input (wrong value, empty sequence, unknown name).
    #[error("Ungültige Eingabe: {0}")]
    InvalidInput(String),
    /// Call in the wrong state.
    #[error("Ungültiger Zustand: {0}")]
    InvalidState(String),
    /// Position or range outside the sequence.
    #[error("Außerhalb des gültigen Bereichs: {0}")]
    OutOfRange(String),
    /// Error when reading or writing files.
    #[error("Dateifehler: {0}")]
    Io(String),
    /// An operation failed (message for the UI).
    #[error("{0}")]
    Operation(String),
}

impl From<std::io::Error> for Error {
    fn from(e: std::io::Error) -> Self {
        Error::Io(e.to_string())
    }
}

/// Result with the core error type.
pub type Result<T> = std::result::Result<T, Error>;
```

```rust
    #[test]
    fn zeilen_bei_laenge_125_und_60_pro_zeile() {
        let l = SequenceLayout::new(125, 60).unwrap();
        assert_eq!(l.line_count(), 3);
```
